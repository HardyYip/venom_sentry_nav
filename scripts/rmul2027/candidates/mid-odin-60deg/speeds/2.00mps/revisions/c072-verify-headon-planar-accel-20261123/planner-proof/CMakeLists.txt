cmake_minimum_required(VERSION 3.16)
project(c072_grid_recovery_proof)
find_package(ament_cmake REQUIRED)
set(dependencies lifecycle_msgs nav2_core nav2_navfn_planner nav2_costmap_2d pluginlib
  rclcpp rclcpp_lifecycle tf2_ros)
foreach(dependency IN LISTS dependencies)
  find_package(${dependency} REQUIRED)
endforeach()
add_executable(grid-proof proof.cpp)
target_compile_features(grid-proof PRIVATE cxx_std_17)
target_include_directories(grid-proof PRIVATE ../source/rmul_c072_planner/src)
ament_target_dependencies(grid-proof ${dependencies})
install(TARGETS grid-proof RUNTIME DESTINATION lib/${PROJECT_NAME})
find_package(behaviortree_cpp_v3 REQUIRED)
add_executable(tree-proof tree-proof.cpp)
target_compile_features(tree-proof PRIVATE cxx_std_17)
ament_target_dependencies(tree-proof behaviortree_cpp_v3)
install(TARGETS tree-proof RUNTIME DESTINATION lib/${PROJECT_NAME})
add_executable(clearance-proof clearance-proof.cpp)
target_compile_features(clearance-proof PRIVATE cxx_std_17)
target_include_directories(clearance-proof PRIVATE ../source/rmul_c072_planner/src)
ament_target_dependencies(clearance-proof nav2_costmap_2d)
install(TARGETS clearance-proof RUNTIME DESTINATION lib/${PROJECT_NAME})
find_package(dwb_core REQUIRED)
find_package(nav2_util REQUIRED)
add_executable(sampling-proof sampling-proof.cpp)
target_compile_features(sampling-proof PRIVATE cxx_std_17)
ament_target_dependencies(sampling-proof dwb_core nav2_util pluginlib)
install(TARGETS sampling-proof RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
