cmake_minimum_required(VERSION 3.16)
project(c063_nav2_probes)
find_package(ament_cmake REQUIRED)
foreach(dependency nav2_costmap_2d nav2_util tf2_ros sensor_msgs dwb_plugins nav_2d_msgs dwb_msgs geometry_msgs)
  find_package(${dependency} REQUIRED)
endforeach()
add_executable(observation_probe observation.cpp)
ament_target_dependencies(observation_probe nav2_costmap_2d nav2_util tf2_ros sensor_msgs)
add_executable(trajectory_probe trajectory.cpp)
ament_target_dependencies(trajectory_probe dwb_plugins nav2_util nav_2d_msgs dwb_msgs geometry_msgs)
enable_testing()
add_test(NAME installed_observation_buffer COMMAND observation_probe)
add_test(NAME installed_dwb_trajectory_integrator COMMAND trajectory_probe)

add_executable(footprint_probe footprint.cpp)
ament_target_dependencies(footprint_probe nav2_costmap_2d geometry_msgs)
add_test(NAME installed_footprint_raster_margin COMMAND footprint_probe)
