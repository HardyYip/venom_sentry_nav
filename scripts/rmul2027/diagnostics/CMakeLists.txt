cmake_minimum_required(VERSION 3.10)
project(timing_header_observer)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(IGN REQUIRED ignition-transport11 ignition-msgs8)
add_executable(timing_chain_observer timing_header_observer.cpp)
target_include_directories(timing_chain_observer PRIVATE ${IGN_INCLUDE_DIRS})
target_link_directories(timing_chain_observer PRIVATE ${IGN_LIBRARY_DIRS})
target_link_libraries(timing_chain_observer ${IGN_LIBRARIES})
ament_target_dependencies(timing_chain_observer rclcpp rcl_interfaces rosgraph_msgs nav_msgs sensor_msgs tf2_msgs)
