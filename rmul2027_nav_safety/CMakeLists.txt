cmake_minimum_required(VERSION 3.16)
project(rmul2027_nav_safety)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rosgraph_msgs geometry_msgs tf2_ros behaviortree_cpp_v3)
  find_package(${dep} REQUIRED)
endforeach()
add_library(rmul_pose_fresh_condition_bt_node SHARED src/pose_fresh.cpp)
target_compile_definitions(rmul_pose_fresh_condition_bt_node PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(rmul_pose_fresh_condition_bt_node rclcpp rosgraph_msgs tf2_ros behaviortree_cpp_v3)
add_executable(clock_probe tests/probe.cpp)
ament_target_dependencies(clock_probe rclcpp rosgraph_msgs geometry_msgs tf2_ros behaviortree_cpp_v3)
install(TARGETS rmul_pose_fresh_condition_bt_node LIBRARY DESTINATION lib)
ament_package()
