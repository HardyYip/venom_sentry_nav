cmake_minimum_required(VERSION 3.16)
project(rmul2027_decision)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
foreach(dep builtin_interfaces rclcpp rclcpp_action action_msgs nav2_msgs nav_msgs geometry_msgs example_interfaces std_msgs std_srvs tf2_ros ament_index_cpp)
  find_package(${dep} REQUIRED)
endforeach()
find_package(behaviortree_cpp REQUIRED CONFIG)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/MatchState.msg" DEPENDENCIES builtin_interfaces)
add_executable(decision_node src/decision_node.cpp)
target_include_directories(decision_node PRIVATE include)
target_link_libraries(decision_node BT::behaviortree_cpp)
ament_target_dependencies(decision_node rclcpp rclcpp_action action_msgs nav2_msgs nav_msgs geometry_msgs example_interfaces std_msgs std_srvs tf2_ros ament_index_cpp)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(decision_node "${cpp_typesupport_target}")
add_executable(logic_contract tests/logic_contract.cpp)
target_include_directories(logic_contract PRIVATE include)
target_link_libraries(logic_contract BT::behaviortree_cpp)
enable_testing()
add_test(NAME logic_contract COMMAND logic_contract "${CMAKE_CURRENT_SOURCE_DIR}/behavior_trees/rmul_2027.xml")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME recovery_contract COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/recovery_contract.py")
install(TARGETS decision_node logic_contract DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config behavior_trees reference recovery DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(PROGRAMS tools/start_decision.sh DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
add_test(NAME timing_contract COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/timing_contract.py")
add_test(NAME input_contract COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/input_contract.py")
ament_package()
